cmake_minimum_required(VERSION 3.21)
project(CockpitMediaValidation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Svg Multimedia Network)
qt_standard_project_setup()
qt_add_executable(RadioStreamProbe RadioStreamProbe.cpp)
target_link_libraries(RadioStreamProbe PRIVATE Qt6::Gui Qt6::Multimedia Qt6::Network)
qt_add_executable(MediaDesignRender MediaDesignRender.cpp)
target_link_libraries(MediaDesignRender PRIVATE Qt6::Gui Qt6::Svg)
qt_add_executable(VideoPlaybackProbe VideoPlaybackProbe.cpp ../../src/media/MediaSession.cpp ../../src/media/MediaSession.h ../../src/media/MediaState.cpp ../../src/media/MediaState.h ../../src/media/MusicQueue.cpp ../../src/MusicSource.cpp)
target_include_directories(VideoPlaybackProbe PRIVATE ../../src)
target_link_libraries(VideoPlaybackProbe PRIVATE Qt6::Gui Qt6::Multimedia)
if(MSVC)
    target_compile_options(RadioStreamProbe PRIVATE /utf-8)
    target_compile_options(MediaDesignRender PRIVATE /utf-8)
    target_compile_options(VideoPlaybackProbe PRIVATE /utf-8)
endif()
